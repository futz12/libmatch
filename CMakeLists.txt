cmake_minimum_required(VERSION 3.31)
project(libmatch)

set(CMAKE_CXX_STANDARD 20)

# Configure CCache if available
find_program(CCACHE_FOUND ccache)
if(CCACHE_FOUND)
    set_property(GLOBAL PROPERTY RULE_LAUNCH_COMPILE ccache)
    set_property(GLOBAL PROPERTY RULE_LAUNCH_LINK ccache)
endif(CCACHE_FOUND)

# 导入 MNN
if(CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|amd64")
    set(MNN_AVX512 ON)
    set(MNN_AVX512_VNNI ON)
    set(MNN_AVX2 ON)
    set(MNN_USE_SSE ON)

    set(MNN_ARM82 OFF)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
    set(MNN_AVX512 OFF)
    set(MNN_AVX512_VNNI OFF)
    set(MNN_AVX2 OFF)
    set(MNN_USE_SSE ON)

    set(MNN_ARM82 ON)
else()
    message(FATAL_ERROR "Unsupported architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()

set(MNN_BUILD_PROTOBUFFER       OFF)
set(MNN_BUILD_SHARED_LIBS       OFF)
set(MNN_BUILD_TOOLS             OFF)
set(MNN_BUILD_TRAIN             ON )
set(MNN_OPENCL                  ON )
set(MNN_OPENMP                  ON )
set(MNN_SUPPORT_BF16            ON )
set(MNN_SUPPORT_DEPRECATED_OPV2 OFF)
set(MNN_USE_THREAD_POOL         OFF)

add_subdirectory(3rdpart/MNN)

# 导入 OpenCV
set(BUILD_ITT OFF)
set(BUILD_JAVA OFF)
set(BUILD_PACKAGE OFF)
set(BUILD_PERF_TESTS OFF)
set(BUILD_PROTOBUF OFF)
set(BUILD_SHARED_LIBS OFF)
set(BUILD_TESTS OFF)
set(BUILD_opencv_apps OFF)
set(BUILD_opencv_dnn OFF)
set(BUILD_opencv_gapi OFF)
set(BUILD_opencv_java OFF)
set(BUILD_opencv_java_bindings_generator OFF)
set(BUILD_opencv_js_bindings_generator OFF)
set(BUILD_opencv_ml OFF)
set(BUILD_opencv_objc_bindings_generator OFF)
set(BUILD_opencv_objdetect OFF)
set(BUILD_opencv_photo OFF)
set(BUILD_opencv_python3 OFF)
set(BUILD_opencv_python_bindings_generator OFF)
set(BUILD_opencv_python_tests OFF)
set(BUILD_opencv_stitching OFF)
set(BUILD_opencv_ts OFF)
set(BUILD_opencv_video OFF)
set(BUILD_opencv_videoio OFF)
set(WITH_ADE OFF)
set(WITH_AVIF OFF)
set(WITH_EIGEN OFF)
set(WITH_FFMPEG OFF)
set(WITH_FLATBUFFERS OFF)
set(WITH_GSTREAMER OFF)
set(WITH_IMGCODEC_GIF OFF)
set(WITH_IMGCODEC_HDR OFF)
set(WITH_IMGCODEC_PFM OFF)
set(WITH_IMGCODEC_PXM OFF)
set(WITH_IMGCODEC_SUNRASTER OFF)
set(WITH_ITT OFF)
set(WITH_IPP OFF)
set(WITH_JASPER OFF)
set(WITH_LAPACK OFF)
set(WITH_OBSENSOR OFF)
set(WITH_OPENEXR OFF)
set(WITH_PROTOBUF OFF)
set(WITH_PTHREADS_PF OFF)
set(WITH_TIFF OFF)
set(WITH_V4L OFF)
set(WITH_VA OFF)
set(WITH_VA_INTEL OFF)
set(WITH_VTK OFF)
set(WITH_WEBP OFF)
set(WITH_OPENMP ON)

add_subdirectory(3rdpart/opencv)

file(GLOB OpenCV_INCLUDE_DIRS "3rdpart/opencv/modules/*/include")
list(APPEND OpenCV_INCLUDE_DIRS "3rdpart/opencv/include")

add_executable(libmatch main.cpp
        src/init.cpp)

target_include_directories(libmatch PRIVATE
        3rdpart/MNN/include
        ${OpenCV_INCLUDE_DIRS}
)

target_link_libraries(libmatch MNN)
target_link_libraries(libmatch opencv_core)
